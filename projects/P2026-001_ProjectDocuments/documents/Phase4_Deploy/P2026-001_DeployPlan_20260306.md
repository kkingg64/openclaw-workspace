
### 2.2 docker-compose.yml

```yaml
version: '3.8'

services:
  # ============================================
  # Nginx Reverse Proxy (Port 80/443)
  # ============================================
  nginx:
    image: nginx:latest
    container_name: fabio-nginx
    restart: unless-stopped
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./nginx/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./nginx/ssl:/etc/nginx/ssl:ro
      - ./dashboard:/var/www/dashboard:ro
    networks:
      - fabio-network
    depends_on:
      - dashboard
    # Security: Read-only volumes where possible

  # ============================================
  # Dashboard App (React/Vue)
  # ============================================
  dashboard:
    build: ./dashboard
    container_name: fabio-dashboard
    restart: unless-stopped
    environment:
      - NODE_ENV=production
      - API_URL=${API_URL}
    networks:
      - fabio-network
    # 只允許 Nginx 訪問，唔直接對外

  # ============================================
  # Database (PostgreSQL)
  # ============================================
  postgres:
    image: postgres:15-alpine
    container_name: fabio-postgres
    restart: unless-stopped
    environment:
      - POSTGRES_DB=${DB_NAME}
      - POSTGRES_USER=${DB_USER}
      - POSTGRES_PASSWORD=${DB_PASSWORD}
    volumes:
      - ./data/postgres:/var/lib/postgresql/data
    networks:
      - fabio-network
    # Database backup 見維護部分

  # ============================================
  # Certbot (Let's Encrypt SSL)
  # ============================================
  certbot:
    image: certbot/certbot:latest
    container_name: fabio-certbot
    volumes:
      - ./nginx/ssl:/etc/letsencrypt:rw
      - ./nginx/html:/var/www/html:rw
    networks:
      - fabio-network
    # 運行一次即可，唔需要長期運行

networks:
  fabio-network:
    driver: bridge
    ipam:
      config:
        - subnet: 172.25.0.0/16

volumes:
  postgres-data:
```

## 3. Nginx 配置

### 3.1 nginx.conf (主要配置)

```nginx
# ============================================
# Fabio Corp - Nginx Production Config
# ============================================

events {
    worker_connections 1024;
}

http {
    # ============================================
    # Security Headers
    # ============================================
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    add_header Content-Security-Policy "default-src 'self' http: https: data: blob: 'unsafe-inline'" always;

    # ============================================
    # Rate Limiting
    # ============================================
    limit_req_zone $binary_remote_addr zone=api_limit:10m rate=10r/s;
    limit_req_zone $binary_remote_addr zone=login_limit:10m rate=5r/m;

    # ============================================
    # Basic Settings
    # ============================================
    sendfile on;
    tcp_nopush on;
    tcp_nodelay on;
    keepalive_timeout 65;
    types_hash_max_size 2048;

    # Gzip Compression
    gzip on;
    gzip_vary on;
    gzip_min_length 1024;
    gzip_types text/plain text/css application/json application/javascript text/xml application/xml;

    # ============================================
    # Upstream - Dashboard
    # ============================================
    upstream dashboard_backend {
        server dashboard:3000;
    }

    # ============================================
    # HTTP -> HTTPS Redirect
    # ============================================
    server {
        listen 80;
        server_name your-domain.com;

        # Let's Encrypt Challenge
        location /.well-known/acme-challenge/ {
            root /var/www/html;
        }

        # Redirect all to HTTPS
        location / {
            return 301 https://$host$request_uri;
        }
    }

    # ============================================
    # HTTPS Server
    # ============================================
    server {
        listen 443 ssl http2;
        server_name your-domain.com;

        # SSL Configuration
        ssl_certificate /etc/nginx/ssl/fullchain.pem;
        ssl_certificate_key /etc/nginx/ssl/privkey.pem;
        
        # SSL Security
        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
        ssl_prefer_server_ciphers off;
        ssl_session_cache shared:SSL:10m;
        ssl_session_timeout 1d;

        # HSTS (Uncomment after testing)
        # add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

        # ============================================
        # Dashboard Routes
        # ============================================
        root /var/www/dashboard;
        index index.html;

        # Static files (React/Vue build)
        location / {
            try_files $uri $uri/ /index.html;
        }

        # API Proxy with Rate Limiting
        location /api/ {
            limit_req zone=api_limit burst=20 nodelay;
            
            proxy_pass http://dashboard_backend;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection 'upgrade';
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_cache_bypass $http_upgrade;
            
            # Timeout settings
            proxy_connect_timeout 60s;
            proxy_send_timeout 60s;
            proxy_read_timeout 60s;
        }

        # Login rate limiting
        location /api/login {
            limit_req zone=login_limit burst=5 nodelay;
            
            proxy_pass http://dashboard_backend;
            proxy_http_version 1.1;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
        }

        # Health check (no rate limit)
        location /health {
            access_log off;
            return 200 "OK\n";
            add_header Content-Type text/plain;
        }

        # Deny access to hidden files
        location ~ /\. {
            deny all;
        }
    }
}
```

## 4. 安全配置

### 4.1 Firewall Rules (UFW)

```bash
# ============================================
# Firewall Setup - 請跟住以下步驟
# ============================================

# 1. Check UFW status
sudo ufw status

# 2. Default policies (deny all incoming, allow outgoing)
sudo ufw default deny incoming
sudo ufw default allow outgoing

# 3. Allow SSH (change default port later recommended)
sudo ufw allow 22/tcp

# 4. Allow HTTP/HTTPS
sudo ufw allow 80/tcp
sudo ufw allow 443/tcp

# 5. Enable UFW
sudo ufw enable

# 6. Check status
sudo ufw status verbose
```

### 4.2 Docker Network Isolation

```yaml
# docker-compose.yml 入面既 network config
# 已經設定左：

networks:
  fabio-network:
    driver: bridge
    internal: false  # 可訪問外部
    ipam:
      config:
        - subnet: 172.25.0.0/16  # 私有 subnet
```

**安全原則：**
- Nginx 係唯一對外既 service
- Dashboard 同 Postgres 只能用內部 network 溝通
- 唔好將 container ports 直接 mapping 到 host (除左 Nginx 既 80/443)

### 4.3 SSH 加強安全

```bash
# 1. 編輯 SSH config
sudo nano /etc/ssh/sshd_config

# 2. 修改以下設定：
Port 2222                    # 改變 SSH 端口
PermitRootLogin no           # 禁止 root 登入
PasswordAuthentication no    # 禁用密碼登入
PubkeyAuthentication yes     # 只允許 key 登入

# 3. Restart SSH
sudo systemctl restart sshd

# 4. 更新 UFW
sudo ufw allow 2222/tcp
sudo ufw delete allow 22/tcp
```

### 4.4 Fail2Ban (防止暴力破解)

```bash
# 1. Install Fail2Ban
sudo apt install fail2ban -y

# 2. Copy config
sudo cp /etc/fail2ban/jail.conf /etc/fail2ban/jail.local

# 3. Edit config
sudo nano /etc/fail2ban/jail.local

# 4. 確保有以下設定：
[sshd]
enabled = true
port = 2222
filter = sshd
logpath = /var/log/auth.log
maxretry = 3
bantime = 3600

# 5. Restart
sudo systemctl restart fail2ban
```

## 5. 部署步驟

### 5.1 準備工作

```bash
# 1. SSH 到 VPS
ssh -i ~/.ssh/your_key user@your-vps-ip -p 2222

# 2. Update system
sudo apt update && sudo apt upgrade -y

# 3. Install Docker
curl -fsSL https://get.docker.com | sh
sudo usermod -aG docker $USER
sudo systemctl enable docker
sudo systemctl start docker

# 4. Install Docker Compose
sudo apt install docker-compose -y
# 或者
sudo curl -L "https://github.com/docker/compose/releases/latest/download/docker-compose-$(uname -s)-$(uname -m)" -o /usr/local/bin/docker-compose
sudo chmod +x /usr/local/bin/docker-compose

# 5. Setup Firewall (見上文 4.1)
sudo ufw allow 80/tcp
sudo ufw allow 443/tcp
sudo ufw allow 2222/tcp
sudo ufw enable
```

### 5.2 建立專案目錄

```bash
# 1. Create project directory
sudo mkdir -p /opt/ai-fabio-corp
cd /opt/ai-fabio-corp

# 2. Create subdirectories
sudo mkdir -p nginx/ssl nginx/html dashboard data/postgres data/backups scripts

# 3. Set ownership (replace 'ubuntu' with your user)
sudo chown -R $USER:$USER /opt/ai-fabio-corp
```

### 5.3 建立必要檔案

```bash
# 1. Create .env file
cat > .env << 'EOF'
DB_NAME=fabio_dashboard
DB_USER=fabio_admin
DB_PASSWORD=$(openssl rand -base64 32)
API_URL=http://dashboard:3000
DOMAIN=your-domain.com
EMAIL=your-email@gmail.com
JWT_SECRET=$(openssl rand -base64 32)
EOF

# 2. Create docker-compose.yml (copy from section 2.2)
nano docker-compose.yml

# 3. Create nginx.conf (copy from section 3.1)
nano nginx/nginx.conf

# 4. Create Dockerfile for dashboard
cat > dashboard/Dockerfile << 'EOF'
FROM node:20-alpine as build
WORKDIR /app
COPY package*.json ./
RUN npm ci --only=production
COPY . .
RUN npm run build

FROM nginx:alpine
COPY --from=build /app/dist /var/www/dashboard
EOF
```

### 5.4 SSL 憑證 (Let's Encrypt)

```bash
# 1. 確保 port 80 可以訪問
# 2. 運行 Certbot (需要停止 Nginx)

# Stop nginx first
docker-compose stop nginx

# Run certbot
docker-compose run --rm certbot certonly --webroot \
  --webroot-path=/var/www/html \
  --email your-email@gmail.com \
  --agree-tos \
  --no-eff-email \
  -d your-domain.com

# 3. Check certificates
ls -la nginx/ssl/live/your-domain.com/

# 4. Start nginx
docker-compose up -d nginx
```

### 5.5 第一次 Deploy

```bash
# 1. Build and start all services
docker-compose up -d

# 2. Check logs
docker-compose logs -f

# 3. Check status
docker-compose ps

# 4. Test health
curl https://your-domain.com/health
```

## 6. 維護

### 6.1 Backup 策略

#### 自動 Backup Script

```bash
#!/bin/bash
# scripts/backup.sh

# ============================================
# Fabio Corp - Automated Backup Script
# ============================================

BACKUP_DIR="/opt/ai-fabio-corp/data/backups"
DATE=$(date +%Y%m%d_%H%M%S)
DB_NAME="fabio_dashboard"
DB_USER="fabio_admin"

# Create backup directory if not exists
mkdir -p $BACKUP_DIR

# ============================================
# Database Backup
# ============================================
echo "[$(date)] Starting database backup..."
docker-compose exec -T postgres pg_dump -U $DB_USER $DB_NAME > $BACKUP_DIR/db_$DATE.sql
gzip $BACKUP_DIR/db_$DATE.sql

# ============================================
# Keep only last 7 days
# ============================================
find $BACKUP_DIR -name "db_*.sql.gz" -mtime +7 -delete

echo "[$(date)] Backup completed: db_$DATE.sql.gz"

# ============================================
# Upload to cloud (optional)
# ============================================
# Add your cloud upload script here (AWS S3, Google Drive, etc.)
```

```bash
# Add to crontab (每日凌晨 3 點)
crontab -e

# Add this line:
0 3 * * * /opt/ai-fabio-corp/scripts/backup.sh >> /var/log/fabio-backup.log 2>&1
```

### 6.2 Update 程序

```bash
#!/bin/bash
# scripts/update.sh

# ============================================
# Fabio Corp - Update Script
# ============================================

echo "[$(date)] Starting update..."

docker-compose pull

docker-compose down

docker image prune -f

```
